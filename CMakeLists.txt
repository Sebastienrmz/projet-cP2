cmake_minimum_required(VERSION 3.16)
project(markov_graph C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(markov_graph src/main.c src/graph.c)
target_include_directories(markov_graph PRIVATE src)
if(MSVC)
    target_compile_definitions(markov_graph PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(markov_graph PRIVATE /W4)
else()
    target_compile_options(markov_graph PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(markov_graph PRIVATE m)
endif()
